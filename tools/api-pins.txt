# Expected answers for tools/check-api.sh, one contract per line.
#
#   pin <probe> = <value>        every host that answers the probe must print exactly this
#   na  <probe> <host>...        these hosts may print `n/a (<reason>)` for the probe: the
#                                situation cannot be posed in their type system. Any other
#                                host printing n/a, or a listed host answering with a value
#                                that is not the pin, fails. A reason is mandatory.
#
# Agreement between hosts is not enough (all of them can share a bug); a pin is what the
# spec says. The T12 family (docs/interim/2026-09-29/worklist/tests/12-integration.md)
# added the flow-sensitive dependencies() contract (spec/SPEC.md §8), Program reuse,
# non-source / non-native input as E_BAD_ARG, and the host-function argument reader.

pin program.deps.aggregate-body-assign = A L
pin program.deps.cond-with-default-assigns = X Y
pin program.deps.assign-in-argument = -

pin program.reuse.after-error = E_DIV_ZERO 1:3|t"2"|E_DIV_ZERO 1:3
